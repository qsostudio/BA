@app.post("/api/informe-pdf")
async def informe_pdf(req: ParcelaRequest, user=Depends(get_current_user)):
    """
    Genera el informe PDF completo de la parcela: reutiliza EXACTAMENTE
    los mismos cálculos que /analizar-parcela, /vaguadas, /punto-pozo-profundo
    y la geología (mismas funciones, mismos parámetros), para que el PDF
    explique con fidelidad lo que ya se ve en la app — no un cálculo aparte.
    """
    if not GEE_OK:
        raise HTTPException(status_code=503, detail="Google Earth Engine no disponible")
    try:
        resultado_gee      = analizar_parcela_gee(req.coords)
        resultado_vaguadas = await calcular_vaguadas(req.coords)
        resultado_pozo     = await encontrar_punto_pozo_profundo(req.coords)

        lats = [c[1] for c in req.coords]
        lons = [c[0] for c in req.coords]
        lat_c, lon_c = sum(lats) / len(lats), sum(lons) / len(lons)
        geo = await obtener_geologia(lat_c, lon_c)
        roca_es, _, estado_geo = clasificar_geologia(geo.get("lith", ""))
        geologia = {
            "roca": roca_es, "nombre_unidad": geo.get("name", "Desconocido"),
            "descripcion": geo.get("descrip", "Sin descripción"), "estado": estado_geo,
        }

        pdf_bytes = generar_informe_pdf(req.coords, resultado_gee, resultado_vaguadas, geologia, resultado_pozo)
        return StreamingResponse(
            BytesIO(pdf_bytes),
            media_type="application/pdf",
            headers={"Content-Disposition": "attachment; filename=informe_agua_subterranea.pdf"},
        )
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

    29	def _fmt(v, decimales=2, sufijo=""):
    30	    """Formatea un número con seguridad, o ' - ' si es None."""
    31	    if v is None:
    32	        return " - "
    33	    try:
    34	        return f"{float(v):.{decimales}f}{sufijo}"
    35	    except (TypeError, ValueError):
    36	        return str(v)

    39	def _limpiar_texto(texto) -> str:
    40	    """
    41	    Sustituye caracteres Unicode fuera de Latin-1 por equivalentes seguros.
    42	    Necesario porque datos que vienen de fuentes externas (nombres
    43	    geológicos del IGME/Macrostrat, etiquetas de ign_service.py) pueden
    44	    traer rayas, comillas tipográficas u otros símbolos que la fuente
    45	    básica de fpdf2 no soporta y harían fallar la generación del PDF.
    46	    """
    47	    if not isinstance(texto, str):
    48	        texto = str(texto)
    49	    reemplazos = {
    50	        "—": " - ", "–": "-", "…": "...", "•": "-",
    51	        "⚠": "[!]", "→": "->", "≥": ">=", "≤": "<=",
    52	        "\u2018": "'", "\u2019": "'", "\u201c": '"', "\u201d": '"',
    53	    }
    54	    for viejo, nuevo in reemplazos.items():
    55	        texto = texto.replace(viejo, nuevo)
    56	    # Red de seguridad final: cualquier carácter que siga sin caber en
    57	    # Latin-1 (emojis, símbolos raros) se elimina en vez de romper el PDF.
    58	    return texto.encode("latin-1", "ignore").decode("latin-1")

   352	    pdf.parrafo(
   353	        "Para cada una de las 400 celdas, se comparan sus 8 vecinas (N, NE, E, SE, S, SO, O, NO) y "
   354	        "se elige aquella hacia la que la pendiente es más pronunciada cuesta abajo: "
   355	        "pendiente = (altura_celda - altura_vecina) / distancia."
   356	    )
   357	    pdf.subtitulo("Paso 2  -  Acumulación de flujo")
   358	    pdf.parrafo(
   359	        "Se recorren las celdas de mayor a menor altura (orden topológico) y cada celda suma a su "
   360	        "vecina de destino su propia acumulación  -  así, una celda con acumulación = 50 significa que "
   361	        "50 celdas de la malla terminan drenando hacia ahí según el relieve."
   362	    )
   363	    pdf.subtitulo("Paso 3  -  Clasificación en escorrentía / vaguada")
   364	    pdf.parrafo(
   365	        f"Una celda se considera parte de la red de drenaje si su acumulación >= {8} (umbral "
   366	        f"STREAM_THR). Dentro de la red de drenaje, se distingue:\n"
   367	        f"  -  Escorrentía (flujo difuso): acumulación entre 8 y 24 celdas.\n"
   368	        f"  -  Vaguada (cauce concentrado): acumulación >= 25 celdas (umbral VAGUADA_THR).\n"
   369	        "Ambos umbrales son heurísticos, ajustables, no una norma oficial."
   370	    )
   371	    pdf.subtitulo("Paso 4  -  Caudal de cada tramo (Método Racional)")
   372	    pdf.parrafo(
   373	        "Q (m³/s) = C · I · A / 360,  con A en hectáreas e I en mm/h.\n"
   374	        "El área de drenaje A se calcula multiplicando la acumulación de la celda por el área real "
   375	        "que representa cada celda de la malla (calculada a partir del espaciado en metros). "
   376	        "C = 0,30 (coeficiente de escorrentía típico de terreno agrícola/rural) e I = 20 mm/h "
   377	        "(lluvia de referencia moderada-intensa) son valores de referencia ASUMIDOS, no medidos en "
   378	        "esta parcela  -  el caudal resultante es orientativo."
   379	    )
   380	
   381	    lineas = [f for f in vaguadas.get("geojson", {}).get("features", [])
   382	              if f["properties"].get("tipo") == "linea_drenaje"]
   383	    lineas_top = sorted(lineas, key=lambda f: f["properties"].get("caudal_ls", 0), reverse=True)[:15]
   384	
   385	    if lineas_top:
   386	        pdf.subtitulo(f"Tramos con mayor caudal estimado (mostrando {len(lineas_top)} de {len(lineas)} totales)")
   387	        with pdf.table(col_widths=(30, 30, 30, 22, 26, 40),
   388	                        text_align=("CENTER", "CENTER", "CENTER", "CENTER", "CENTER", "LEFT")) as table:
   389	            row = table.row()
   390	            for h in ["Tipo", "Latitud", "Longitud", "Acum.", "Área (ha)", "Caudal (L/s)"]:
   391	                row.cell(h, style=FontFace(emphasis="BOLD"))
   392	            for f in lineas_top:
   393	                p = f["properties"]
   394	                lon0, lat0 = f["geometry"]["coordinates"][0]
   395	                row = table.row()
   396	                row.cell("Vaguada" if p["categoria"] == "vaguada" else "Escorrentía")
   397	                row.cell(f"{lat0:.5f}")
   398	                row.cell(f"{lon0:.5f}")
   399	                row.cell(_fmt(p.get("acumulacion"), 1))
   400	                row.cell(_fmt(p.get("area_ha"), 3))
   401	                row.cell(_fmt(p.get("caudal_ls"), 2))
   402	    else:
   403	        pdf.parrafo("No se ha detectado ningún tramo de drenaje por encima del umbral en esta parcela.")
   404	
   405	    pdf.ln(2)
   406	    pdf.subtitulo("Resumen de la red de drenaje")
   407	    pdf.parrafo(
   408	        f"Resolución de la malla: {vaguadas.get('resolucion', ' - ')}\n"
   409	        f"Tramos totales: {vaguadas.get('n_lineas', 0)}  (de ellos, {vaguadas.get('n_vaguadas', 0)} son vaguadas)\n"
   410	        f"Acumulación máxima registrada: {_fmt(vaguadas.get('acum_maxima'), 1)} celdas\n"
   411	        f"Caudal máximo estimado en la parcela: {_fmt(vaguadas.get('caudal_maximo_ls'), 2)} L/s"
   412	    )

   415	def _seccion_puntos_optimos(pdf: InformePDF, vaguadas: dict):
   416	    pdf.add_page()
   417	    pdf.titulo_seccion(5, "Puntos óptimos para balsa o pozo")
   418	    pdf.parrafo(
   419	        "De las 400 celdas de la malla, se seleccionan las de mayor acumulación de flujo  -  son los "
   420	        "puntos donde el relieve concentra más agua de escorrentía, y por tanto los mejores "
   421	        "candidatos para una balsa de recogida o para situar un pozo."
   422	    )
   423	    pdf.dato_consultado(
   424	        "La misma malla de acumulación de flujo D8 calculada en la sección anterior.",
   425	        "Se ordenan las 400 celdas de mayor a menor acumulación y se toman hasta 7, exigiendo que "
   426	        "cada una esté separada de las ya elegidas al menos 2 celdas (para no agrupar varios puntos "
   427	        "prácticamente en el mismo sitio)."
   428	    )
   429	
   430	    puntos = [f for f in vaguadas.get("geojson", {}).get("features", [])
   431	              if f["properties"].get("tipo") == "punto_acumulacion"]
   432	    puntos = sorted(puntos, key=lambda f: f["properties"].get("acumulacion", 0), reverse=True)
   433	
   434	    if puntos:
   435	        with pdf.table(
   436	            col_widths=(14, 40, 40, 30, 54),
   437	            text_align=("CENTER", "CENTER", "CENTER", "CENTER", "LEFT"),
   438	        ) as table:
   439	            row = table.row()
   440	            for h in ["#", "Latitud", "Longitud", "Acumulación", "Interpretación"]:
   441	                row.cell(h, style=FontFace(emphasis="BOLD"))
   442	            for i, f in enumerate(puntos, start=1):
   443	                p = f["properties"]
   444	                lon0, lat0 = f["geometry"]["coordinates"]
   445	                row = table.row()
   446	                row.cell(str(i))
   447	                row.cell(f"{lat0:.6f}")
   448	                row.cell(f"{lon0:.6f}")
   449	                row.cell(_fmt(p.get("acumulacion"), 1))
   450	                etiqueta = p.get("etiqueta", "").replace("\U0001F4A7 ", "")
   451	                row.cell(etiqueta)
   452	    else:
   453	        pdf.parrafo("No se ha detectado ningún punto de acumulación significativo en esta parcela.")
   454

   508	def generar_informe_pdf(coords: list, resultado_gee: dict, resultado_vaguadas: dict, geologia: dict, resultado_pozo: dict = None) -> bytes:
   509	    """Genera el informe completo y devuelve los bytes del PDF."""
   510	    pdf = InformePDF()
   511	    fecha_txt = datetime.now().strftime("%d/%m/%Y %H:%M")
   512	
   513	    _portada(pdf, coords, fecha_txt)
   514	    _seccion_resumen(pdf, resultado_gee, resultado_vaguadas, geologia)
   515	    _seccion_punto_optimo(pdf, resultado_gee)
   516	    _seccion_punto_pozo(pdf, resultado_pozo)
   517	    _seccion_d8(pdf, resultado_vaguadas)
   518	    _seccion_puntos_optimos(pdf, resultado_vaguadas)
   519	    _seccion_geologia(pdf, geologia)
   520	    _seccion_fuentes(pdf)
   521	
   522	    return bytes(pdf.output())
